MT6628 DT example for Lenovo B8000-H
====================================

The GPIO numbers below are facts from the Lenovo vendor DCT. GPIO_ACTIVE_HIGH
describes the electrical level of PMU_EN and RST (RST high means run). The
MT6320 RTC 32 kHz output is enabled by the board PMIC setup and has no CCF
provider yet, so the optional clocks property is intentionally absent.

&pio {
	mt6628_uart3_pins: mt6628-uart3-pins {
		pins-rx {
			pinmux = <MT6589_GPIO219__FUNC_URXD3>;
			input-enable;
			bias-pull-up;
		};

		pins-tx {
			pinmux = <MT6589_GPIO220__FUNC_UTXD3>;
		};
	};
};

&uart3 {
	status = "okay";
	pinctrl-names = "default";
	pinctrl-0 = <&uart3_pins>;

	mt6628 {
		compatible = "mediatek,mt6628-wmt";

		pmu-en-gpios = <&pio 14 GPIO_ACTIVE_HIGH>;
		reset-gpios = <&pio 225 GPIO_ACTIVE_HIGH>;

		/* BGF: GPIO26 / EINT8, level low. */
		interrupt-parent = <&pio>;
		interrupts = <8 IRQ_TYPE_LEVEL_LOW>;

		mediatek,mmc-host = <&mmc3>;
		mediatek,operational-speed = <2000000>;
		mediatek,rtc-stable-ms = <100>;
		mediatek,off-stable-ms = <10>;
		mediatek,reset-stable-ms = <30>;
		mediatek,on-stable-ms = <100>;
	};
};

&mmc3 {
	status = "okay";
	bus-width = <4>;
	max-frequency = <50000000>;
	non-removable;
	cap-sdio-irq;
	keep-power-in-suspend;
	/delete-property/ vmmc-supply;
	/delete-property/ mmc-pwrseq;

	/* Do not copy the blade.dtsi vemc supplies: they belong to eMMC. */
	mt6628_wifi: wifi@1 {
		compatible = "mediatek,mt6628-wifi";
		reg = <1>;
		/* Native SDIO IRQ is preferred.  Vendor fallback was GPIO128 /
		 * EINT9, active low; the current driver does not request both.
		 */
	};
};
