 * Internal-storage profile.  The existing Android USRDATA logical partition
 * is /dev/mmcblk0p20; the partition table and identity partitions stay intact.
 */

#include "mt6589-lenovo-b8000-h.dts"

/ {
	model = "Lenovo YOGA Tablet 10 B8000-H (eMMC root)";

	chosen {
		bootargs = "clk_ignore_unused printk.always_kmsg_dump=1 rdinit=/init console=tty0 loglevel=8 ignore_loglevel root=/dev/mmcblk0p20";
	};
};

&mmc0 {
	status = "okay";

	card@0 {
		partitions {
			/* USRDATA is the Linux root; every other partition stays read-only. */
			partition@58780000 {
				/delete-property/ read-only;
			};
		};
	};
};

	status = "disabled";
